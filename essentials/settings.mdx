Every Mintlify site needs a `docs.json` file with the core configuration settings. Learn more about the [properties](#properties) below.

The keys below are properties the current Mintlify `docs.json` schema accepts. The schema sets `additionalProperties` to false, so older names such as `topbarLinks`, `topbarCtaButton`, `footerSocials`, `modeToggle`, and `backgroundImage` are rejected.

## Properties

<ResponseField name="name" type="string" required>
Name of your project. Used for the global title.

Example: `mintlify`

</ResponseField>

<ResponseField name="navigation" type="object" required>
  The navigation structure. This value is an object. It is not an array of groups.

  Use one content branch on that object: `groups`, `pages`, `tabs`, `anchors`, `versions`, `languages`, `dropdowns`, or `products`. `global` may sit beside the branch. Do not put `tabs`, `anchors`, or `versions` at the root of `docs.json`.

  This repo uses the `tabs` branch plus `global`. See [Navigation](/essentials/navigation) for nested pages.

  <Expandable title="Tabs branch">
    <ResponseField name="tabs" type="object[]">
      Each tab is an object. `tab` is the label. `groups` holds the pages for that tab.

      <Expandable title="Tab">
        <ResponseField name="tab" type="string" required>
          The tab label.

          Example: `Guides`
        </ResponseField>
        <ResponseField name="groups" type="object[]">
          Groups inside the tab.

          <Expandable title="Group">
            <ResponseField name="group" type="string" required>
              The name of the group.

              Example: `Getting started`
            </ResponseField>
            <ResponseField name="pages" type="array">
              Page paths without the `.mdx` extension, or nested group objects.

              Example: `["index", "quickstart", "development"]`
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="global" type="object">
      External links that stay available on every section.

      <Expandable title="Global">
        <ResponseField name="anchors" type="object[]">
          Each anchor needs `anchor` and `href`.

          <Expandable title="Anchor">
            <ResponseField name="anchor" type="string" required>
              The anchor label.

              Example: `Documentation`
            </ResponseField>
            <ResponseField name="href" type="string" required>
              A path or external URL.

              Example: `https://mintlify.com/docs`
            </ResponseField>
            <ResponseField name="icon" type="string">
              Icon name.

              Example: `book-open-cover`
            </ResponseField>
            <ResponseField name="color" type="object">
              Optional hex colors for the anchor. This is not a single hex string.
              <Expandable title="Color">
                <ResponseField name="light" type="string">
                  Hex color used in light mode. Example: `#16A34A`
                </ResponseField>
                <ResponseField name="dark" type="string">
                  Hex color used in dark mode. Example: `#15803D`
                </ResponseField>
              </Expandable>
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>

  `versions` is a different navigation branch, not a root array of strings. Each item is an object with a `version` name. Do not combine `versions` with `tabs` on the same navigation object.

  ```json
  {
    "navigation": {
      "versions": [
        {
          "version": "v1",
          "groups": [
            {
              "group": "Getting started",
              "pages": ["index", "quickstart"]
            }
          ]
        }
      ]
    }
  }
  ```

</ResponseField>

<ResponseField name="logo" type="string or object">
  Path to logo image or object with path to "light" and "dark" mode logo images
  <Expandable title="Logo">
    <ResponseField name="light" type="string">
      Path to the logo in light mode
    </ResponseField>
    <ResponseField name="dark" type="string">
      Path to the logo in dark mode
    </ResponseField>
    <ResponseField name="href" type="string" default="/">
      Where clicking on the logo links you to
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="favicon" type="string">
  Path to the favicon image
</ResponseField>

<ResponseField name="colors" type="object" required>
  Hex color codes for your global theme. The only accepted fields are `primary`, `light`, and `dark`.
  <Expandable title="Colors">
    <ResponseField name="primary" type="string" required>
      The primary color. Used most often for highlighted content, section
      headers, and accents in light mode.
    </ResponseField>
    <ResponseField name="light" type="string">
      The light color. Used as the primary color in dark mode.
    </ResponseField>
    <ResponseField name="dark" type="string">
      The dark color. Used as the primary color in light mode.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="navbar" type="object">
  Links and the primary action in the top bar.
  <Expandable title="Navbar">
    <ResponseField name="links" type="object[]">
      Navbar links. A normal link needs `label` and `href`. Set `type` to `github` or `discord` for those links. `href` is still required.

      Example: `{ "label": "Support", "href": "mailto:hi@mintlify.com" }`
    </ResponseField>
    <ResponseField name="primary" type="object">
      The primary call to action. `type` is `button`, `github`, or `discord`. `href` is required. `label` is required when `type` is `button`.

      Example: `{ "type": "button", "label": "Dashboard", "href": "https://dashboard.mintlify.com" }`
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="api" type="object">
  Configuration for API pages and the API playground.
  <Expandable title="API">
    <ResponseField name="openapi" type="string | object | array">
      A path or URL to an OpenAPI file, an array of those values, or an object with a `source` path. This field belongs under `api`, not at the root of `docs.json`.

      <CodeGroup>
        ```json Path
        "openapi": "/api-reference/openapi.json"
        ```
        ```json URL
        "openapi": "https://example.com/openapi.json"
        ```
        ```json Multiple
        "openapi": ["/api-reference/openapi.json", "https://example.com/openapi.json"]
        ```
      </CodeGroup>
    </ResponseField>

    <ResponseField name="mdx" type="object">
      Settings for API pages written in MDX.
      <Expandable title="MDX">
        <ResponseField name="server" type="string | string[]">
          Base URL for MDX API pages. Use an array to offer more than one base URL.

          Example: `https://example.com`
        </ResponseField>
        <ResponseField name="auth" type="object">
          Authentication for MDX API pages.
          <Expandable title="Auth">
            <ResponseField name="method" type="string">
              One of `bearer`, `basic`, `key`, or `cobo`.
            </ResponseField>
            <ResponseField name="name" type="string">
              Authentication name for the API.

              Example: `Authorization`
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="playground" type="object">
      Configurations for the API playground.
      <Expandable title="Playground">
        <ResponseField name="display" type="string" default="interactive">
          The display mode of the API playground. One of `interactive`, `simple`, `none`, or `auth`. Defaults to `interactive`. Use `auth` to show the playground only to authenticated users.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="footer" type="object">
  Footer content. Social profiles live under `socials`, not a root `footerSocials` object.
  <Expandable title="Footer">
    <ResponseField name="socials" type="object">
      Each key is a platform and each value is a profile URL. Accepted keys are `x`, `website`, `facebook`, `youtube`, `discord`, `slack`, `github`, `linkedin`, `instagram`, `hacker-news`, `medium`, `telegram`, `twitter`, `x-twitter`, `earth-americas`, `bluesky`, `threads`, `reddit`, and `podcast`.

      Example:

      ```json
      {
        "x": "https://x.com/mintlify",
        "github": "https://github.com/mintlify",
        "linkedin": "https://linkedin.com/company/mintlify"
      }
      ```
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="appearance" type="object">
  Light and dark mode settings.
  <Expandable title="Appearance">
    <ResponseField name="default" type="string" default="system">
      One of `system`, `light`, or `dark`.
    </ResponseField>
    <ResponseField name="strict" type="boolean" default="false">
      Set to `true` to hide the light and dark mode toggle.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="background" type="object">
  Background image, decoration, and color. This replaces a root `backgroundImage` string and `colors.background`.
  <Expandable title="Background">
    <ResponseField name="image" type="string or object">
      A path or URL used for both modes, or an object with `light` and `dark` paths.
    </ResponseField>
    <ResponseField name="decoration" type="string">
      One of `gradient`, `grid`, or `windows`.
    </ResponseField>
    <ResponseField name="color" type="object">
      Hex colors for the page background.
      <Expandable title="Color">
        <ResponseField name="light" type="string">
          Background color in light mode. Example: `#FFFFFF`
        </ResponseField>
        <ResponseField name="dark" type="string">
          Background color in dark mode. Example: `#0B0F0C`
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
